private _knowledgeGraph: WorkspaceKnowledgeGraph | null = null;  // + _decisionGraph, _taskGraph, _evolutionHistory, _dna
private readonly organsBase: string;                             // = config.organsBasePath ?? join(root_path, '.uccp', 'persist')
get knowledgeGraph(): WorkspaceKnowledgeGraph; get decisionGraph(): WorkspaceDecisionGraph;
get taskGraph(): WorkspaceTaskGraph; get evolutionHistory(): WorkspaceEvolutionHistory; get dna(): WorkspaceDNA;

private dnaInputs(): Record<string, unknown> {
  const byStatus: Record<string, number> = {};
  for (const d of this.worldModel.decisions) byStatus[d.status] = (byStatus[d.status] ?? 0) + 1;
  return { genome: this.genome, standards: this.worldModel.standards, decisions: { count: this.worldModel.decisions.length, byStatus } };
}

async persistWorkspace(): Promise<void> {
  const base = join(this.organsBase);
  this.dna.recompute(this.dnaInputs());
  await this.knowledgeGraph.persist(join(base, 'knowledge-graph.json'));
  await this.decisionGraph.persist(join(base, 'decision-graph.json'));
  await this.taskGraph.persist(join(base, 'task-graph.json'));
  await this.evolutionHistory.persist(join(base, 'evolution-history.json'));
  await this.dna.persist(join(base, 'workspace-dna.json'));
}
async loadWorkspace(): Promise<number> {
  const base = join(this.organsBase);
  let restored = 0;
  restored += await this.knowledgeGraph.load(join(base, 'knowledge-graph.json'));
  restored += await this.decisionGraph.load(join(base, 'decision-graph.json'));
  restored += await this.taskGraph.load(join(base, 'task-graph.json'));
  restored += await this.evolutionHistory.load(join(base, 'evolution-history.json'));
  restored += await this.dna.load(join(base, 'workspace-dna.json'));
  return restored;
}
close(): void {
  this._knowledgeGraph?.close(); this._decisionGraph?.close();
  this._taskGraph?.close(); this._evolutionHistory?.close();
  this._knowledgeGraph = null; this._decisionGraph = null;

'')` (organism.ts:466-487 convention); empty-path/empty-message guards skip ingestion; payload paths treated as opaque strings — handlers never touch the filesystem with them |
| T-01-02 | Information Disclosure | secrets/personal data in graph properties or DNA inputs | mitigate | Handlers whitelist exactly the documented payload fields (path/hash/message/files/language/lines/title/url/number/pr_number); never store env vars or credentials; DNA hashes only (one-way sha256, never reversible) |
| T-01-03 | DoS | unbounded graph growth from event spam | mitigate | Deterministic upsert ids (`file:${path}`, `commit:${hash}`, `build:${event.id}` — occurrence ids from bus-assigned event.id, never crypto.randomUUID() in handlers) + organ-internal caps (maxNodes/maxCycles/maxMutations, Plan 01-02) |
| T-01-04 | Tampering | SQLite + manifest integrity under .uccp/persist | mitigate | Per-organ GraphStore isolation; lazy creation defers I/O until real use; tests close() before rmSync (universal-harness fix in Task 2); manifests via writeSnapshot (mkdir-recursive, atomic single-write JSON) |
| T-01-05 | Spoofing | event source field | accept | Local single-user bus; provenance labels are fixed ('system_log'); matches existing organism trust posture |
