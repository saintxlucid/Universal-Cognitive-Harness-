```ts
export type HormoneName = 'dopamine' | 'cortisol' | 'melatonin' | 'adrenaline' | 'serotonin';
export type HormoneLevels = Record<HormoneName, number>;
export type OperatingMode = 'explore' | 'caution' | 'maintain' | 'sprint' | 'stabilize' | 'balanced';

export interface HormoneInput {
  signals: GlobalSignals;       // the existing 8-signal bundle
  recentRewards: number;        // 0-1 decaying reward counter
  recentPains: number;          // 0-1 decaying pain counter
  stability: number;            // 0-1: 1 = clean streak (merges landing, no errors)
  night: boolean;               // circadian phase === 'night'
}

export function computeHormones(input: HormoneInput): HormoneLevels;   // PURE
export function modeForLevels(h: HormoneLevels): OperatingMode;        // PURE
```

- Canonical mapping (THE tested contract, RESEARCH <specifics>):
  - `dopamine = clamp(0.55*recentRewards + 0.30*userSatisfactionTrend + 0.15*confidence)`
  - `cortisol = clamp(0.50*recentPains + 0.35*uncertainty + 0.15*riskLevel)`
  - `melatonin = night ? clamp(0.60 + 0.40*(1 - urgency)) : 0.10`
  - `adrenaline = clamp(0.70*urgency + 0.30*resourcePressure)`
  - `serotonin = clamp(0.50*stability + 0.30*userSatisfactionTrend - 0.20*technicalDebtLevel)`
  - `clamp(x) = Math.max(0, Math.min(1, x))`. law:17 fusion — every hormone
    blends ≥ 2 independent signals; no hormone reads a single signal.
- Mode: the strictly-dominant hormone wins; tie-break order
  `[cortisol, adrenaline, melatonin, dopamine, serotonin]`; when ALL levels

## Homeostatic Regulator (src/physiology/homeostasis.ts — new)

```ts
export type RegulationAction = 'prune' | 'consolidate' | 'throttle' | 'heal' | 'grow' | 'evolve' | 'escalate';
export interface HomeostaticMetric { id: string; value: number; target: number; tolerance: number; }
export interface RegulationDecision { metric: string; deviation: number; action: RegulationAction; }
export interface RegulationReport { decisions: RegulationDecision[]; executed: RegulationAction[]; skipped: RegulationAction[]; at: Date; }

export interface HomeostasisAdapters {
  throttle?: (metric: HomeostaticMetric) => void | Promise<void>;   // Metabolism BudgetOverride
  consolidate?: (metric: HomeostaticMetric) => void | Promise<void>; // SleepCycle nap()
  heal?: (metric: HomeostaticMetric) => void | Promise<void>;        // connectome integrity repair
  escalate?: (metric: HomeostaticMetric) => void | Promise<void>;    // executive/nervous alert
  evolve?: (metric: HomeostaticMetric) => void | Promise<void>;      // gene expression modulation
}

export class HomeostaticRegulator {
  constructor(config: { clock: Clock; adapters?: HomeostasisAdapters; maxActionsPerCycle?: number; eventSink?: (event: { type: 'homeostasis:regulated'; payload: { actions: RegulationAction[]; metrics: HomeostaticMetric[] } }) => void });
  registerSensor(id: string, read: () => number, target: number, tolerance: number): void;
  registerAction(action: RegulationAction, executor: (decision: RegulationDecision) => void | Promise<void>): void;  // slots: prune, grow
  async regulate(now?: Date): Promise<RegulationReport>;
  getMetrics(): HomeostaticMetric[];
}
```

- Deviation: `deviation = (value - target) / Math.max(Math.abs(target), 1e-6)`.
  A decision fires when `|deviation| > tolerance` AND the metric's mapped
  action has an executor (adapter or registered slot). Metric→action map
  (DEFAULT_ACTION_MAP, configurable per sensor via `registerSensor`):
  `memoryGrowth→prune`, `tokenUsage→throttle`, `contextSize→consolidate`,
  `cpuLoad→throttle`, `apiCosts→throttle`, `latency→escalate`,
  `knowledgeFreshness→prune`, `confidence→consolidate`, `pluginHealth→escalate`,
  `graphIntegrity→heal`.
- `regulate()` executes at most `maxActionsPerCycle` (default 3) actions,
  sequentially, awaiting executors; actions without an executor land in
  `skipped` (Wave A ships with `prune`/`grow` skipped — Wave B wires them via
  `registerAction`). Never called synchronously inside an event handler —
  the exoskeleton schedules it (circadian-driven).
- Deterministic: same sensors + same clock → same report. Report `at` comes
  from the injected clock.
