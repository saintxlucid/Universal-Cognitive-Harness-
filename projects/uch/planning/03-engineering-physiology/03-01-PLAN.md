```ts
export type ReflexVerdict = 'allow' | 'block' | 'defer';

export interface WriteProposal {
  tool: string;            // 'file-editor' | 'command-runner' | 'diff-review' | other
  target: string;          // file path or command string; zones derive from path.dirname(target)
  content?: string;        // full content for writes, diff text for diff-review
  zone?: string;           // optional explicit zone override
}

export interface ReflexEvidence {
  check_id: string;
  rule: string;            // human-readable rule name
  level: 'block' | 'defer'; // what this check's finding means
  evidence: string[];      // node ids, thresholds hit, identifiers — MUST be concrete
  threshold?: string;      // e.g. 'overlap>=0.6'
  reason: string;
}

export interface ReflexCheckContext {
  zoneAffect(zone: string): number; // pain-memory seam; default () => 0
}

export interface ReflexCheck {
  id: string;              // kebab-case, e.g. 'duplicate-abstraction'
  rule: string;
  evaluate(proposal: WriteProposal, ctx: ReflexCheckContext): ReflexEvidence | null; // null = pass
}

export interface ReflexResult {
  verdict: ReflexVerdict;
  checks_run: string[];    // check ids evaluated
  evidence: ReflexEvidence[];
  override?: { name: string; recorded: boolean }; // set when the escape hatch suppressed a block
  duration_ms: number;
  target: string;
}

export type ZoneAffectProvider = (zone: string) => number;

export interface ReflexGateConfig {
  checks?: ReflexCheck[];
  zoneAffect?: ZoneAffectProvider;               // default () => 0 (L1 seam — pain memory lands in 03-04)
  override?: { enabled?: boolean; names?: string[] }; // escape hatch; env UCH_REFLEX_OVERRIDE also honored
  signalSink?: (signal: Signal) => void;         // default none; sync fire-and-forget wrapper in wiring
  eventSink?: (event: Omit<NeuralEvent, 'id' | 'timestamp'>) => void; // default none
}

export interface ReflexGateLike {
  evaluate(proposal: WriteProposal): ReflexResult;
}
```

## Gate semantics (src/reflex/gate.ts — Plan 03-02 creates)
- `new ReflexGate(config?: ReflexGateConfig)`; `register(check: ReflexCheck): void` (appends to registry); `getChecks(): ReflexCheck[]`.
- `evaluate(proposal: WriteProposal): ReflexResult` — **SYNC** (no Promise anywhere in the signature). Deterministic: same proposal + same state → same verdict, repeatedly.
- Verdict rules:
  - Any check returning evidence with `level: 'block'` → `block` (first block wins; later evidence still collected into `evidence[]`).
  - Else any `level: 'defer'` → `defer`.
  - Else `allow`.
  - **Pain escalation (D-08 seam):** if `ctx.zoneAffect(zoneOf(proposal)) >= 0.7` (exported constant `PAIN_ESCALATION_THRESHOLD = 0.7`), every `defer` evidence in the affected zone escalates to `block`. `zoneOf` = `proposal.zone ?? path.dirname(proposal.target)` (for targets without a path separator, zone = target).
- Verdict id/evidence determinism: evidence entries carry the check's own `check_id`; result-level ids are NOT separate — assertions check `evidence[0]!.check_id === 'duplicate-abstraction'` and evidence strings contain concrete identifiers (node ids like `file:src/legacy/user-manager.ts`, threshold text).
- **Escape hatch (D-02):** `override.names` (or env `UCH_REFLEX_OVERRIDE`, comma-separated; `'*'` = all) suppresses `block` evidence for the named checks — the gate then returns `allow` but attaches `override: { name: <check id>, recorded: true }` AND still invokes `eventSink` with the block record including the override field (violations recorded, never silently skipped). A suppressed `defer` downgrades to allow with the same override record.
- **Recording (D-02):** when `signalSink`/`eventSink` are provided:
  - `block` → eventSink receives `{ type: 'governance:event_denied', source: 'reflex-gate', payload: { gate: 'reflex', tool, target, check_id, rule, evidence, threshold, reason, override } }`; signalSink receives `createSignal('error:occurred', 'reflex-gate', { ...payload, verdict: 'block' }, { interrupt: true, interruptLevel: 2 })` (L5: error:occurred maps to brainstem priority 2).
  - `defer` → eventSink receives `{ type: 'review:requested', source: 'reflex-gate', payload: { tool, target, checks: <ids>, reason } }`; signalSink receives `createSignal('review:requested', 'reflex-gate', { ...payload, verdict: 'defer' }, { interrupt: false })`.
  - `allow` → eventSink receives `{ type: 'tool:called', source: 'reflex-gate', payload: { tool, target, verdict: 'allow', checks_run, duration_ms } }`; no signal.
  - **NO new event types** (D-04): these three types only. Event records are fire-and-forget — `evaluate()` must not await anything.
- `zoneAffect` is read once per evaluate and passed to every check via ctx.
